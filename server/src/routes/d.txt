2. Ticket generation — "Team Chat" click hone par API route
typescript// zeva-clinic/app/api/messenger/create-ticket/route.ts
import { NextRequest, NextResponse } from "next/server";
import crypto from "crypto";
import redis from "@/lib/redis";
import { getServerSession } from "@/lib/auth"; // tera existing Zeva auth check

const TICKET_EXPIRY_SECONDS = 60; // 1 minute - one-time use, short-lived

export async function POST(req: NextRequest) {
  try {
    // Step 1: confirm user is logged into Zeva Clinic (existing session check)
    const session = await getServerSession(req);
    if (!session || !session.user) {
      return NextResponse.json({ error: "Not authenticated" }, { status: 401 });
    }

    const { user } = session;

    // Step 2: generate a random, unguessable one-time ticket
    const ticket = crypto.randomBytes(32).toString("hex");

    // Step 3: store ticket -> user payload in Redis, with short TTL
    const payload = {
      zevaUserId: user.id,
      clinicId: user.clinicId,
      name: user.name,
      avatarUrl: user.avatarUrl || null,
      role: user.role, // "doctor" | "receptionist" | "staff" | "admin"
    };

    await redis.set(
      `sso_ticket:${ticket}`,
      JSON.stringify(payload),
      "EX",
      TICKET_EXPIRY_SECONDS
    );

    // Step 4: return the messenger URL with the ticket
    const messengerUrl = `${process.env.MESSENGER_URL}/auth/sso?ticket=${ticket}`;

    return NextResponse.json({ redirectUrl: messengerUrl });
  } catch (err) {
    console.error("Failed to create SSO ticket", err);
    return NextResponse.json({ error: "Failed to create ticket" }, { status: 500 });
  }
}

"use client";

import { useState } from "react";
import { MessageSquare, Loader2 } from "lucide-react";

const TeamChatButton = () => {
  const [isLoading, setIsLoading] = useState(false);

  const handleClick = async () => {
    setIsLoading(true);
    try {
      const res = await fetch("/api/messenger/create-ticket", {
        method: "POST",
        credentials: "include", // Zeva ka session cookie bhejne ke liye
      });

      if (!res.ok) throw new Error("Failed to create ticket");

      const { redirectUrl } = await res.json();

      // naye tab me kholna ya same tab me redirect - business decision hai
      window.open(redirectUrl, "_blank"); // naya tab
      // ya: window.location.href = redirectUrl;  // same tab
    } catch (err) {
      console.error(err);
      alert("Could not open Team Chat. Please try again.");
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <button
      onClick={handleClick}
      disabled={isLoading}
      className="flex items-center gap-2 rounded-lg bg-primary px-4 py-2 text-sm font-medium text-primary-foreground hover:bg-primary/90 disabled:opacity-50"
    >
      {isLoading ? (
        <Loader2 className="h-4 w-4 animate-spin" />
      ) : (
        <MessageSquare className="h-4 w-4" />
      )}
      Team Chat
    </button>
  );
};

4. Internal verify-ticket endpoint — Zeva Clinic side (jo Messenger backend call karega)
Yeh wahi endpoint hai jo tera Messenger ka auth.service.ts (verifyTicketWithZeva) call karta hai:
typescript// zeva-clinic/app/api/internal/verify-ticket/route.ts
import { NextRequest, NextResponse } from "next/server";
import redis from "@/lib/redis";

export async function POST(req: NextRequest) {
  try {
    // Step 1: verify request is genuinely from Messenger (shared secret)
    const apiKey = req.headers.get("x-internal-api-key");
    if (apiKey !== process.env.MESSENGER_INTERNAL_API_KEY) {
      return NextResponse.json({ error: "Forbidden" }, { status: 403 });
    }

    const { ticket } = await req.json();
    if (!ticket) {
      return NextResponse.json({ error: "Ticket is required" }, { status: 400 });
    }

    // Step 2: look up ticket in Redis
    const key = `sso_ticket:${ticket}`;
    const data = await redis.get(key);

    if (!data) {
      return NextResponse.json(
        { error: "Invalid or expired ticket" },
        { status: 401 }
      );
    }

    // Step 3: CRITICAL - delete ticket immediately (one-time use only)
    await redis.del(key);

    const payload = JSON.parse(data);

    return NextResponse.json(payload);
  } catch (err) {
    console.error("Ticket verification failed", err);
    return NextResponse.json({ error: "Verification failed" }, { status: 500 });
  }
}
.env (Zeva Clinic side, iske liye):

Yeh exact wahi value honi chahiye jo Messenger ke .env me ZEVA_INTERNAL_API_KEY hai — dono taraf same secret.

Kyun Zeva Clinic hi ticket banayega
Ticket ka purpose hai: "main confirm kar raha hoon ki yeh user abhi Zeva Clinic me genuinely logged in hai." Yeh confirmation sirf wahi system de sakta hai jiske paas actual session/login information hai — matlab Zeva Clinic khud, kyunki wahi user ka original login state maintain karta hai (session cookie, JWT, jo bhi Zeva apna existing auth use karta hai).
Messenger ke paas yeh jaankari hai hi nahi ki user Zeva me login hai ya nahi — Messenger toh ek bilkul separate service hai jise abhi tak pata hi nahi ki yeh user kaun hai. Isliye Messenger khud se ticket generate nahi kar sakta — usko verify karna padta hai ki ticket Zeva ne banaya tha aur woh valid hai.
Analogy se samajh
Socho tu kisi building me ja raha hai jahan reception (Zeva Clinic) tujhe pehle verify karta hai (tera ID card check karke), phir ek visitor pass (ticket) deta hai. Doosri floor ka security guard (Messenger) tera face nahi pehchanta, wo sirf yeh check karta hai ki tere paas reception ka diya hua valid pass hai ya nahi. Guard khud visitor pass nahi bana sakta — kyunki usko pata hi nahi tu kaun hai, wo sirf pass ko verify kar sakta hai jo reception se aaya.

ZEVA CLINIC (ticket ISSUE karta hai)
├── User "Team Chat" click karta hai
├── Zeva confirm karta hai: "haan yeh user mera hi logged-in staff hai"
├── Zeva ek random ticket banata hai (crypto.randomBytes)
├── Redis me store: ticket -> { zevaUserId, clinicId, name, role }
└── Browser ko redirect URL deta hai: messenger.zeva.com/auth/sso?ticket=xyz

                    ↓ (browser redirect ke through ticket Messenger tak pahunchta hai)

ZEVA MESSENGER (ticket VERIFY karta hai)
├── Frontend ticket ko URL se leta hai
├── Backend ko bhejta hai: "yeh ticket check karo"
├── Messenger backend Zeva Clinic ko server-to-server call karta hai
│   ("ye ticket tumne banaya tha? valid hai?")
├── Zeva confirm karta hai + user data deta hai + ticket delete karta hai
└── Messenger ab apna session bana deta hai (access + refresh token)
Code-level mapping (jo maine pehle diya)

create-ticket endpoint → yeh Zeva Clinic ke codebase me hai (maine zeva-clinic/app/api/messenger/create-ticket/route.ts diya tha)
verify-ticket endpoint → yeh bhi Zeva Clinic ke codebase me hai (zeva-clinic/app/api/internal/verify-ticket/route.ts) — Messenger isko call karta hai, khud nahi banata
auth.service.ts ka verifyTicketWithZeva → yeh Messenger ke codebase me hai, jo Zeva ke verify-ticket endpoint ko hit karta hai

Toh simple rule yaad rakh: jo system user ki identity confirm kar sakta hai (Zeva), wahi ticket issue karta hai. Jo system identity confirm nahi kar sakta (Messenger), wo sirf verify karta hai.
